比较简单，直接做
```tsx
  function sumAndMultiply(n: number): number {
        // 转成字符串，遍历去掉 0
        const str = String(n)

        // 相加
        let sum = 0
        let x = ""
        for(let i = 0; i < str.length ;i++){
          if(str[i]!=="0"){
            sum += Number(str[i])
            x += str[i]
          }
        }

        return sum * Number(x)
    };
```
